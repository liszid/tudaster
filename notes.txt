#-------------------------------------------------------------------------#

For adding new file, use the precompiled {sampleFiles} at php/SampleFiles

#-------------------------------------------------------------------------#

Page canonical is built in &x=MainPage&y=SubPage&z=Action order

#-------------------------------------------------------------------------#

Links are categorized by functionality.
    data-link : Tells which page to be loaded
    data-post : Sends specific post data (mostly ids)
    data-toggle : implies special action (modal, dropdown)

Functionality is autocompiled from 05_ajx.events.js

#-------------------------------------------------------------------------#

Javascript and CSS files must be loaded orderwise, which is set by default

#-------------------------------------------------------------------------#

For adding new [Hardware] subcategory, use the sample below.

./php/Page/Hardware.php
#   Add line in $urlPaths as follows
    ,"Subcategory" => array("path" => "Hardware/Subcategory", "role" => "canHardware")

./php/Page_Handlers/sHardware.php
#   Add lines in const TYPES as follows
    ,"Subcategory" => array(
        "origo" => 'usrHardwSubcategory'
        ,"optionalData" => self::NETWORKING //Or any fitting preset
        ,"tableName" => 'Hardware_Subcategory'
    )

./php/Samples/sTables/HARDWARE.php
#   Add a constant for its Table generating
    const HARDWARE_SUBCATEGORY = array(
        "tableId" => "tableHardwareSubcategory"
        ,"tableRoot" => "Hardware/Subcategory"
        ,"tableRole" => "canHardware"
        ,"data" => self::HARDWARE_NETWORKING //Or any fitting preset
        ,"button" => self::HARDWARE_BUTTONS
    );

./php/Samples/sTables/HARDWARE.php
#   Add line to TABLES_BY_NAMES as follows
    ,"Hardware_Subcategory" => self::HARDWARE_SUBCATEGORY
